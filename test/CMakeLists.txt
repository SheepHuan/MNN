set(TEST_DEPS ${MNN_DEPS})
if(MNN_WITH_PLUGIN)
  include(${CMAKE_CURRENT_LIST_DIR}/plugin/CMakeLists.txt)
  list(APPEND TEST_DEPS plugin_matmul)
endif()

if (CMAKE_SYSTEM_NAME MATCHES "^Android")
  list(APPEND TEST_DEPS android)
endif()

if(APPLE)
  file(GLOB_RECURSE Files ${CMAKE_CURRENT_LIST_DIR}/*.cpp ${CMAKE_CURRENT_LIST_DIR}/*.mm)
else()
  file(GLOB_RECURSE Files ${CMAKE_CURRENT_LIST_DIR}/*.cpp)
endif()

if(NOT MNN_BUILD_TRAIN)
  file(GLOB_RECURSE GradTestFiles ${CMAKE_CURRENT_LIST_DIR}/grad/*.cpp)
  file(GLOB_RECURSE NNTestFiles ${CMAKE_CURRENT_LIST_DIR}/nn/*.cpp)
  list(REMOVE_ITEM Files ${GradTestFiles} ${NNTestFiles})
endif()

if(NOT MNN_CUDA)
  file(GLOB_RECURSE CudaBenchOpFiles ${CMAKE_CURRENT_LIST_DIR}/bench_ops/cuda/*.cpp)
  list(REMOVE_ITEM Files ${CudaBenchOpFiles})
endif()

if(NOT MNN_OPENCL)
  file(GLOB_RECURSE OpenCLBenchOpFiles ${CMAKE_CURRENT_LIST_DIR}/bench_ops/opencl/*.cpp)
  list(REMOVE_ITEM Files ${OpenCLBenchOpFiles})
endif()

set(OPENCL_WRAPPER_RUNTIME_FILE
  ${CMAKE_CURRENT_LIST_DIR}/../source/backend/opencl/core/runtime/OpenCLWrapper.cpp)
set(RUN_TEST_OPENCL_EXTRA_SOURCES)
set(RUN_TEST_OPENCL_USE_WRAPPER OFF)

if(MNN_OPENCL)
  if(TARGET MNN_CL)
    list(APPEND TEST_DEPS MNN_CL)
  endif()
  if(OpenCL_LIBRARIES)
    list(APPEND TEST_DEPS ${OpenCL_LIBRARIES})
  elseif(MNN_SEP_BUILD)
    list(APPEND RUN_TEST_OPENCL_EXTRA_SOURCES ${OPENCL_WRAPPER_RUNTIME_FILE})
    set(RUN_TEST_OPENCL_USE_WRAPPER ON)
  endif()
endif()

add_executable(run_test.out ${Files} ${RUN_TEST_OPENCL_EXTRA_SOURCES})
if(MNN_CUDA)
  find_package(CUDA 8.0 REQUIRED)
  target_include_directories(run_test.out PRIVATE ${CUDA_INCLUDE_DIRS})
  target_link_libraries(run_test.out ${CUDA_LIBRARIES})
  if(CUDA_CUBLAS_LIBRARIES)
    target_link_libraries(run_test.out ${CUDA_CUBLAS_LIBRARIES})
  else()
    find_library(CUDA_CUBLAS_LIBRARY cublas
      HINTS
        ${CUDA_TOOLKIT_ROOT_DIR}/lib64
        ${CUDA_TOOLKIT_ROOT_DIR}/lib
        ${CUDA_TOOLKIT_ROOT_DIR}/targets/aarch64-linux/lib)
    if(CUDA_CUBLAS_LIBRARY)
      target_link_libraries(run_test.out ${CUDA_CUBLAS_LIBRARY})
    else()
      target_link_libraries(run_test.out -lcublas)
    endif()
  endif()
endif()
target_link_libraries(run_test.out ${MNN_DEPS})
if(RUN_TEST_OPENCL_USE_WRAPPER)
  target_compile_definitions(run_test.out PRIVATE MNN_USE_LIB_WRAPPER)
  if(CMAKE_SYSTEM_NAME MATCHES "Linux")
    target_link_libraries(run_test.out dl)
  endif()
endif()
if (MSVC)
  target_compile_options(run_test.out PRIVATE /bigobj)
endif()
if (MNN_SUPPORT_BF16)
    target_compile_options(run_test.out PRIVATE -DMNN_SUPPORT_BF16)
endif()
if (NOT MNN_BUILD_SHARED_LIBS)
  if(APPLE)
    set(TEST_DEPS -Wl,-all_load ${TEST_DEPS} -Wl,-noall_load)
  elseif(CMAKE_CXX_COMPILER_ID MATCHES "GNU" OR CMAKE_CXX_COMPILER_ID MATCHES "Clang")
    set(TEST_DEPS -Wl,--whole-archive ${TEST_DEPS} -Wl,--no-whole-archive)
  elseif(MSVC)
    foreach (DEPEND ${TEST_DEPS})
      target_link_options(run_test.out PRIVATE /WHOLEARCHIVE:$<TARGET_FILE:${DEPEND}>)
    endforeach ()
  endif()
else()
  target_link_libraries(run_test.out ${TEST_DEPS})
endif()
target_include_directories(run_test.out PRIVATE
    ${CMAKE_CURRENT_LIST_DIR}/
    ${CMAKE_CURRENT_LIST_DIR}/../transformers/llm/engine/src/)
if (APPLE)
  find_library(FOUNDATION Foundation REQUIRED)
  target_link_libraries(run_test.out ${FOUNDATION})
endif()
